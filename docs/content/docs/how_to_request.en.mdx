### HTTP Request

<Accordions type="single">
  <Accordion title="How to get api token?">
    You can get API token at [panel](https://fragapi.com/dashboard/api-tokens)

    Enter name and expiry date of the api token.

  </Accordion>
  <Accordion title="How to make requests?">
    Requests are made to `https://api.fragapi.com/v1/name-of-the-method`
 	
    Also required to pass the authorization header, with API token,
    Like: `Authorization`: `Bearer YourAPI_TOKENfromPanel`
  </Accordion>

### Examples using programming languages

```python tab="example.py"
import requests

BASE_URL = "https://api.fragapi.com/v1"
token = "<Ваш API ключ>"

r = requests.get(
  url=f"{BASE_URL}/users/me",
  headers={
    "Authorization": f"Bearer {token}",
    "Content-Type": "application/json"
  }
)
me = r.json()

# Should print info about the balance, etc..
print(me)
```

```java tab="example.java"
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;

public class Example {
    public static void main(String[] args) throws Exception {
        String baseUrl = "https://api.fragapi.com/v1";
        String token = "<Ваш API ключ>";

        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(baseUrl + "/users/me"))
                .header("Authorization", "Bearer " + token)
                .header("Content-Type", "application/json")
                .GET()
                .build();

        HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());
        String me = response.body();

        // Should print info about balance, etc...
        System.out.println(me);
    }
}
```

```go tab="example.go"
package main

import (
	"fmt"
	"io"
	"net/http"
)

func main() {
	baseURL := "https://api.fragapi.com/v1"
	token := "<Ваш API ключ>"

	req, _ := http.NewRequest("GET", baseURL+"/users/me", nil)
	req.Header.Set("Authorization", "Bearer "+token)
	req.Header.Set("Content-Type", "application/json")

	client := &http.Client{}
	resp, _ := client.Do(req)
	defer resp.Body.Close()

	body, _ := io.ReadAll(resp.Body)
	me := string(body)

	// Should print info about balance, etc...
	fmt.Println(me)
}
```

```rust tab="example.rs"
use reqwest;

#[tokio::main]
async fn main() -> Result<(), reqwest::Error> {
    let base_url = "https://api.fragapi.com/v1";
    let token = "<Ваш API ключ>";

    let client = reqwest::Client::new();
    let me = client
        .get(format!("{}/users/me", base_url))
        .header("Authorization", format!("Bearer {}", token))
        .header("Content-Type", "application/json")
        .send()
        .await?
        .json::<serde_json::Value>()
        .await?;

    // Should print info about balance, etc...
    println!("{:?}", me);

    Ok(())
}
```
