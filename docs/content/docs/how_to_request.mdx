### Примеры на языках программирования

```python tab="<PythonIcon /> example.py"
import requests

BASE_URL = "https://api.fragapi.com/v1"
token = "<Ваш API ключ>"

# Делаем запрос к API передавая свой API ключ
r = requests.get(
  url=f"{BASE_URL}/users/me",
  headers={
    "Authorization": f"Bearer {token}",
    "Content-Type": "application/json"
  }
)
me = r.json()

# Должно вывести данные о балансе и тд..
print(me)
```

```javascript tab="<JavascriptIcon /> example.js"
const baseUrl = "https://api.fragapi.com/v1";
const token = "<Ваш API ключ>";

// Делаем запрос к API передавая свой API ключ
const response = await fetch(`${baseUrl}/users/me`, {
  method: "GET",
  headers: {
    Authorization: `Bearer ${token}`,
    "Content-Type": "application/json",
  },
});

const me = await response.json();

// Должно вывести данные о балансе и тд..
console.log(me);
```

```go tab="<GoIcon /> example.go"
package main

import (
	"fmt"
	"io"
	"net/http"
)

func main() {
	baseURL := "https://api.fragapi.com/v1"
	token := "<Ваш API ключ>"

	req, _ := http.NewRequest("GET", baseURL+"/users/me", nil)
	req.Header.Set("Authorization", "Bearer "+token)
	req.Header.Set("Content-Type", "application/json")

	client := &http.Client{}
	resp, _ := client.Do(req)
	defer resp.Body.Close()

	body, _ := io.ReadAll(resp.Body)
	me := string(body)

	// Должно вывести данные о балансе и тд..
	fmt.Println(me)
}
```

```rust tab="<RustIcon /> example.rs"
use reqwest;

#[tokio::main]
async fn main() -> Result<(), reqwest::Error> {
    let base_url = "https://api.fragapi.com/v1";
    let token = "<Ваш API ключ>";

    let client = reqwest::Client::new();
    let me = client
        .get(format!("{}/users/me", base_url))
        .header("Authorization", format!("Bearer {}", token))
        .header("Content-Type", "application/json")
        .send()
        .await?
        .json::<serde_json::Value>()
        .await?;

    // Должно вывести данные о балансе и тд..
    println!("{:?}", me);

    Ok(())
}
```

```java tab="<JavaIcon /> example.java"
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;

public class Example {
    public static void main(String[] args) throws Exception {
        String baseUrl = "https://api.fragapi.com/v1";
        String token = "<Ваш API ключ>";

        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(baseUrl + "/users/me"))
                .header("Authorization", "Bearer " + token)
                .header("Content-Type", "application/json")
                .GET()
                .build();

        HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());
        String me = response.body();

        // Должно вывести данные о балансе и тд..
        System.out.println(me);
    }
}
```

### Пример POST запроса

Данные передаются в теле запроса в формате JSON. Пример покупки 50 звезд через [`POST /v1/stars/buy`](/ru/docs/api/stars/buy_stars):

<Callout type="warn">
  Это настоящая покупка: звезды будут отправлены пользователю, а сумма списана с вашего баланса.
</Callout>

```python tab="<PythonIcon /> buy.py"
import requests

BASE_URL = "https://api.fragapi.com/v1"
token = "<Ваш API ключ>"

# Данные покупки передаются в теле запроса
r = requests.post(
  url=f"{BASE_URL}/stars/buy",
  headers={"Authorization": f"Bearer {token}"},
  json={"username": "synca", "quantity": 50}
)

# При ошибке приходит {"error": ..., "detail": ...} и статус 4xx
if not r.ok:
  print(r.status_code, r.json())
else:
  print(r.json())
```

```javascript tab="<JavascriptIcon /> buy.js"
const baseUrl = "https://api.fragapi.com/v1";
const token = "<Ваш API ключ>";

// Данные покупки передаются в теле запроса
const response = await fetch(`${baseUrl}/stars/buy`, {
  method: "POST",
  headers: {
    Authorization: `Bearer ${token}`,
    "Content-Type": "application/json",
  },
  body: JSON.stringify({ username: "synca", quantity: 50 }),
});

// При ошибке приходит {"error": ..., "detail": ...} и статус 4xx
console.log(response.status, await response.json());
```

```go tab="<GoIcon /> buy.go"
package main

import (
	"fmt"
	"io"
	"net/http"
	"strings"
)

func main() {
	baseURL := "https://api.fragapi.com/v1"
	token := "<Ваш API ключ>"

	// Данные покупки передаются в теле запроса
	body := strings.NewReader(`{"username": "synca", "quantity": 50}`)

	req, _ := http.NewRequest("POST", baseURL+"/stars/buy", body)
	req.Header.Set("Authorization", "Bearer "+token)
	req.Header.Set("Content-Type", "application/json")

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		panic(err)
	}
	defer resp.Body.Close()

	// При ошибке приходит {"error": ..., "detail": ...} и статус 4xx
	data, _ := io.ReadAll(resp.Body)
	fmt.Println(resp.StatusCode, string(data))
}
```

```rust tab="<RustIcon /> buy.rs"
use reqwest;
use serde_json::json;

#[tokio::main]
async fn main() -> Result<(), reqwest::Error> {
    let base_url = "https://api.fragapi.com/v1";
    let token = "<Ваш API ключ>";

    // Данные покупки передаются в теле запроса
    let response = reqwest::Client::new()
        .post(format!("{}/stars/buy", base_url))
        .header("Authorization", format!("Bearer {}", token))
        .json(&json!({ "username": "synca", "quantity": 50 }))
        .send()
        .await?;

    // При ошибке приходит {"error": ..., "detail": ...} и статус 4xx
    let status = response.status();
    let data = response.json::<serde_json::Value>().await?;
    println!("{} {:?}", status, data);

    Ok(())
}
```

```java tab="<JavaIcon /> Buy.java"
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;

public class Buy {
    public static void main(String[] args) throws Exception {
        String baseUrl = "https://api.fragapi.com/v1";
        String token = "<Ваш API ключ>";

        // Данные покупки передаются в теле запроса
        String body = "{\"username\": \"synca\", \"quantity\": 50}";

        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(baseUrl + "/stars/buy"))
                .header("Authorization", "Bearer " + token)
                .header("Content-Type", "application/json")
                .POST(HttpRequest.BodyPublishers.ofString(body))
                .build();

        HttpResponse<String> response = HttpClient.newHttpClient()
                .send(request, HttpResponse.BodyHandlers.ofString());

        // При ошибке приходит {"error": ..., "detail": ...} и статус 4xx
        System.out.println(response.statusCode() + " " + response.body());
    }
}
```
